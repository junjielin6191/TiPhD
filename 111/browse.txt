//--------------------Browse页面所需的函数
document.addEventListener('DOMContentLoaded', () => {
    const celltypeButton = document.getElementById('celltype-button');
    const spatialButton = document.getElementById('spatial-button');
    const tableHeader = document.getElementById('table-header');
    const tableBody = document.getElementById('table-body');

    const loadData = (type) => {
        fetch(`/browse/data/${type}`)
            .then((response) => {
                if (!response.ok) {
                    throw new Error(`Failed to load data: ${response.statusText}`);
                }
                return response.json();
            })
            .then((data) => {
                console.log("Data received from server:", data);
    
                // 定义字段顺序
                const fieldOrder = type === "spatiallayer"
                    ? ["SLID","species", "tissue_class", "tissue_type", "cancer_type",  "spatial_layer", "Cell_type_composition", "PMID", "Paper_Title", "journal", "year", "technology_type_for_discovery", "technology_platform_for_discovery", "Phenotype_type", "Phenotype_label", "Phenotype_evidence", "model_type", "technology_type_for_validation", "technology_platform_for_validation", "evidence_type"]
                    : ["CTID","species", "tissue_class", "tissue_type", "cancer_type",  "major_cell_type", "cell_name", "Symbol", "PMID", "Paper_Title", "journal", "year", "technology_type_for_discovery", "technology_platform_for_discovery", "Phenotype_type", "Phenotype_label", "Phenotype_evidence", "model_type", "technology_type_for_validation", "technology_platform_for_validation", "evidence_type"];
    
                // 清空表头和表体
                tableHeader.innerHTML = '';
                tableBody.innerHTML = '';
    
                // 生成表头
                fieldOrder.forEach((field) => {
                    const th = document.createElement("th");
                    th.textContent = field;
                    tableHeader.appendChild(th);
                });
    
                // 生成表体
                data.forEach((row) => {
                    const tr = document.createElement("tr");
                    fieldOrder.forEach((field) => {
                        const td = document.createElement("td");
                        td.textContent = row[field] || ""; // 如果字段不存在则显示为空
                        tr.appendChild(td);
                    });
                    tableBody.appendChild(tr);
                });
            })
            .catch((error) => {
                console.error(error);
                alert('Failed to load data. Please try again.');
            });
    };
    

    // 绑定按钮事件
    celltypeButton.addEventListener('click', () => {
        loadData('celltype');
    });

    spatialButton.addEventListener('click', () => {
        loadData('spatiallayer');
    });
});



@app.route("/browse/data/<table_name>")
def get_browse_data(table_name):
    if table_name == "spatiallayer":
        data = session.query(SpatialLayer).all()
        if not data:
            return jsonify({"error": "No data found in spatiallayer"}), 404
        formatted_data = [format_entry(entry, SPATIALLAYER_FIELDS) for entry in data]
        flattened_data = [flatten_entry(item) for item in formatted_data]
        return jsonify(flattened_data)
    elif table_name == "celltype":
        data = session.query(CellType).all()
        if not data:
            return jsonify({"error": "No data found in celltype"}), 404
        formatted_data = [format_entry(entry, CELLTYPE_FIELDS) for entry in data]
        flattened_data = [flatten_entry(item) for item in formatted_data]
        return jsonify(flattened_data)
    return jsonify({"error": "Invalid table name"}), 404
